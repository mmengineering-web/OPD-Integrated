// ---- a Supabase-like client over pg: rpc(fn, args).range(a,b)
let chain = Promise.resolve();
const as = (role, fn) => (chain = chain.then(() => as1(role, fn), () => as1(role, fn)));
const as1 = async (role, fn) => { await db.query("begin"); await db.query("set local role authenticated"); await db.query("select set_config('test.uid',$1,true)", [role]); try { const r = await fn(); await db.query("commit"); return r; } catch (e) { await db.query("rollback"); throw e; } };
const client = (uid) => ({ rpc: (fn, args = {}) => {
  const keys = Object.keys(args); let lo = null, hi = null;
  const run = async () => { try {
    const sql = `select * from public.${fn}(${keys.map((k, i) => `${k} => $${i + 1}${(args[k] !== null && typeof args[k] === "object") ? "::jsonb" : ""}`).join(", ")})` + (lo !== null ? ` offset ${lo} limit ${hi - lo + 1}` : "");
    const res = await as(uid, () => db.query(sql, keys.map(k => (args[k] !== null && typeof args[k] === "object") ? JSON.stringify(args[k]) : args[k])));
    const unwrap = res.fields.length === 1 && res.fields[0].name === fn;
    const data = unwrap ? (res.rows.length === 1 && !["etl_history_read", "etl_shadow_diff"].includes(fn) ? res.rows[0][fn] : res.rows.map(r => r[fn])) : res.rows;
    return { data, error: null };
  } catch (e) { return { data: null, error: { message: e.message } }; } };
  const p = { range: (a, b) => { lo = a; hi = b; return p; }, then: (res, rej) => run().then(res, rej) }; return p; } });

// ---- the worker, shimmed for node (Pyodide from the local copy of the same release)

  importScripts() {}, loadPyodide: () => require("/tmp/pyo/pyodide/pyodide.js").loadPyodide({ indexURL: "/tmp/pyo/pyodide" }), postMessage: (m) => out.push(m) };
